list(GET ${SVARIANT_S} 0 CMP)
list(GET ${VARIANT_ONE} 0 ONE)
list(GET ${VARIANT_TWO} 0 TWO)
if(${CMP} MATCHES ${ONE})
    SET(SVARIANT_S ${VARIANT_TWO})
    set(SELECT_MIKE_VARIANT ${PROJECT_SOURCE_DIR}/.cmake/mike_variant.cmake)
    include(${SELECT_MIKE_VARIANT})
    SET(SVARIANT_S ${VARIANT_ONE})
elseif(${CMP} MATCHES ${TWO})
    SET(SVARIANT_S ${VARIANT_ONE})
    set(SELECT_MIKE_VARIANT ${PROJECT_SOURCE_DIR}/.cmake/mike_variant.cmake)
    include(${SELECT_MIKE_VARIANT})
    SET(SVARIANT_S ${VARIANT_TWO})
endif()
set(SELECT_MIKE_VARIANT ${PROJECT_SOURCE_DIR}/.cmake/mike_variant.cmake)
include(${SELECT_MIKE_VARIANT})
